const REMITENTE = 'joseluispina7@gmail.com';
const EVENTO = 'el-movidon-2026';

function autorizarCorreo() {
  if (Session.getEffectiveUser().getEmail().toLowerCase() !== REMITENTE) {
    throw new Error('Abre el proyecto con la cuenta ' + REMITENTE);
  }
  console.log('Destinatarios disponibles hoy: ' + MailApp.getRemainingDailyQuota());
}

function doPost(event) {
  const raw = event.postData && event.postData.contents;
  if (!raw || raw.length > 30000) return reply({ ok: false, error: 'invalid_request' });
  let request;
  try {
    request = JSON.parse(raw);
  } catch {
    return reply({ ok: false, error: 'invalid_request' });
  }

  const properties = PropertiesService.getScriptProperties();
  const key = properties.getProperty('MOVIDON_MAIL_KEY');
  if (!key || !request || request.key !== key) {
    return reply({ ok: false, error: 'unauthorized' });
  }
  if (Session.getEffectiveUser().getEmail().toLowerCase() !== REMITENTE) {
    return reply({ ok: false, error: 'wrong_sender' });
  }
  if (request.action === 'status') {
    return reply({ ok: true, sender: REMITENTE, drawVersion: 2, quota: MailApp.getRemainingDailyQuota() });
  }

  if (request.action === 'test') {
    if (MailApp.getRemainingDailyQuota() < 1) return reply({ ok: false, error: 'quota' });
    MailApp.sendEmail({
      to: REMITENTE,
      name: 'Amigo invisible El Movidón',
      subject: '[PRUEBA] Tu amigo invisible · El Movidón',
      body: 'El envío desde Gmail funciona. Este mensaje es una prueba y no contiene ninguna asignación real.'
    });
    return reply({ ok: true, sender: REMITENTE });
  }

  if (request.action !== 'send' || !Array.isArray(request.messages) ||
      !request.messages.length || request.messages.length > 20 ||
      !request.messages.every(validAssignment) ||
      (request.drawId !== undefined && !/^[a-f0-9-]{36}$/.test(request.drawId))) {
    return reply({ ok: false, error: 'invalid_request' });
  }

  const lock = LockService.getScriptLock();
  if (!lock.tryLock(5000)) return reply({ ok: false, error: 'busy' });
  try {
    let quota = MailApp.getRemainingDailyQuota();
    const drawKey = request.drawId ? EVENTO + ':' + request.drawId : EVENTO;
    const results = request.messages.map(person => {
      const email = person.email.toLowerCase();
      const digest = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, drawKey + ':' + email);
      const id = 'mail:' + Utilities.base64EncodeWebSafe(digest);
      const state = properties.getProperty(id);
      if (state) return state;
      if (quota < 1) return 'pending';

      properties.setProperty(id, 'review');
      try {
        MailApp.sendEmail({
          to: email,
          name: 'Amigo invisible El Movidón',
          subject: request.drawId ? '[NUEVO SORTEO] Tu amigo invisible · El Movidón' : 'Tu amigo invisible · El Movidón',
          body: [
            '¡' + person.name + '! Ya salió el sorteo 🎅',
            ...(request.drawId ? ['Esta asignación sustituye a la del sorteo anterior.', ''] : []),
            '🎁 Te toca regalarle a ' + person.recipient + '.',
            '📅 Intercambio el 26 de diciembre de 2026.',
            '', '¡Feliz Navidad! 🎄'
          ].join('\n'),
          htmlBody: '<p>¡' + escapeHtml(person.name) + '! Ya salió el sorteo 🎅</p>' +
            (request.drawId ? '<p>Esta asignación sustituye a la del sorteo anterior.</p>' : '') +
            '<p>🎁 Te toca regalarle a <strong>' + escapeHtml(person.recipient) + '</strong>.<br>' +
            '📅 Intercambio el <strong>26 de diciembre de 2026</strong>.</p>' +
            '<p>¡Feliz Navidad! 🎄</p>'
        });
        quota--;
        properties.setProperty(id, 'sent');
        return 'sent';
      } catch {
        quota = 0;
        return 'review';
      }
    });
    return reply({ ok: true, sender: REMITENTE, results });
  } finally {
    lock.releaseLock();
  }
}

function validAssignment(person) {
  return person && typeof person.email === 'string' && person.email.length <= 254 &&
    /^[a-z0-9.!#$%&'*+/=?^_`{|}~-]+@[a-z0-9-]+(?:\.[a-z0-9-]+)+$/i.test(person.email) &&
    typeof person.name === 'string' && person.name.length > 0 && person.name.length <= 80 &&
    typeof person.recipient === 'string' && person.recipient.length > 0 && person.recipient.length <= 80;
}

function escapeHtml(text) {
  const entities = { '&': '&amp;', '<': '&lt;', '>': '&gt;' };
  return text.replace(/[&<>]/g, character => entities[character]);
}

function reply(payload) {
  return ContentService.createTextOutput(JSON.stringify(payload)).setMimeType(ContentService.MimeType.JSON);
}
